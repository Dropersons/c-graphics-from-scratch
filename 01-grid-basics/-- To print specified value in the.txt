-- To print specified value in the borders of a matrix

#include<stdio.h>
#define HEIGHT 10
#define WIDTH 10

int main()
{
    int buffer[HEIGHT][WIDTH    ];
    
    for(int i = 0; i < HEIGHT; i++)
    {
        for(int j = 0; j < WIDTH; j++)
        {
            buffer[i][j] = 0;
        }
    }
    
    for(int i = 0; i < HEIGHT; i++)
    {
        for(int j = 0; j < WIDTH; j++)
        {
            if(i == 0 || i == HEIGHT - 1 || j == 0 || j == WIDTH - 1)
            {
                buffer[i][j] = 1;
            }
        }
    }
    
    for(int i = 0; i < HEIGHT; i++)
    {
        for(int j = 0; j < WIDTH; j++)
        {
            printf("%d" , buffer[i][j]);
        }
        printf("\n");
    }
    return 0;
}
